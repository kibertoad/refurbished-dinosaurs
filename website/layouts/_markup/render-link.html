{{/* Markdown link hook. The site is served from a subdirectory on GitHub Pages, so root-relative destinations written in content ("/games/...") have to be resolved against the baseURL. Hugo's relURL leaves a leading slash alone, hence the trim. */}} {{- $dest := .Destination -}} {{- $external := strings.HasPrefix $dest "http" -}} {{- $bare := or $external (strings.HasPrefix $dest "#") (strings.HasPrefix $dest "mailto:") (strings.HasPrefix $dest "tel:") -}} {{- if not $bare -}} {{- $dest = strings.TrimPrefix "/" $dest | relLangURL -}} {{- end -}} {{- /* A citation of a numbered rule, [STATUS-14](#status-14), has to land on a "###### STATUS-14" heading on the same page (see render-heading.html). Hugo does not check fragments, so a mistyped or dropped number would otherwise link to nothing. */ -}} {{- $rule := upper (strings.TrimPrefix "#" $dest) -}} {{- if and (findRE `^#[a-z]+(-[a-z]+)*-[1-9][0-9]*$` $dest) (eq .PlainText $rule) -}} {{- if not (findRE (printf `(?m)^###### %s[ \t]*$` $rule) .Page.RawContent) -}} {{- errorf "%s: link to rule %s, which this page does not define" .Page.File.Path $rule -}} {{- end -}} {{- end -}} {{- .Text | safeHTML -}} {{- /* This comment removes trailing newlines. */ -}}
